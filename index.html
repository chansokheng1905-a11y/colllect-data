ប្រើប្រាស់ Cloud Firestore Collection តែមួយ

នៅក្នុង App ធ្វើតេស្តគណិតវិទ្យាណាមួយរបស់លោកគ្រូ ពេលសិស្សបំពេញឈ្មោះ រួចឆ្លើយសំណួរចប់ ហើយចុច «បញ្ជូន (Submit)» លោកគ្រូគ្រាន់តែហៅកូដខាងក្រោមនេះ នោះទិន្នន័យនឹងលោតមកកាន់ផ្ទាំងនេះភ្លាមៗ (Real-time)៖

// កូដផ្ញើទិន្នន័យពី App ធ្វើតេស្ត (ដាក់ត្រង់កន្លែង Finish Quiz)
async function sendExamResult() {
await db.collection('exam_hub_submissions').add({
studentName: document.getElementById('studentName').value, // ឈ្មោះសិស្ស
gender: document.getElementById('studentGender').value, // ភេទ (ប្រុស/ស្រី)
gradeLevel: document.getElementById('gradeLevel').value, // ថ្នាក់ (ឧ. 10A, 8C2)
quizTitle: 'តេស្តធរណីមាត្រប្រចាំខែ', // ឈ្មោះតេស្ត
score: userScore, // ពិន្ទុដែលធ្វើបាន (ឧ. 18.5)
maxScore: totalQuestions, // ពិន្ទុអតិបរមា (ឧ. 20)
percentage: ((userScore / totalQuestions) * 100).toFixed(1),
submittedAt: new Date().toISOString()
});
alert('បានបញ្ជូនលទ្ធផលតេស្តជោគជ័យ!');
}
ចំណុចពិសេស៖ App នេះអាចទទួលទិន្នន័យពី App ធ្វើតេស្តច្រើនផ្សេងគ្នាព្រមពេលតែមួយ មិនថា App តេស្តពិជគណិត, ធរណីមាត្រ ឬតេស្តត្រៀមប្រឡងបាក់ឌុបឡើយ!
